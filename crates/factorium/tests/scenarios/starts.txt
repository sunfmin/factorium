// factorium's extension for pi: what the tool adds to a pi
// session that pi takes on no flag. The tool carries this file, keeps it in
// its own user directory for pi, and names it on the line that
// starts the session; nothing else loads it.
//
//   - The guard is asked before every tool call, with the
//     call as pi describes it. The guard, not this file, decides. A call the guard did not allow is
//     refused, with what the guard said; so is one it could not be asked of.
//   - The rules of the worktree, and the added rules, are loaded as pi
//     loads a rules file. pi finds the start directory's own, in the directory it starts
//     in; the worktree and the added directory are others.
//
// It is plain JavaScript and imports nothing of pi's: only what Node has.
import { spawnSync } from "node:child_process"
import { existsSync, readFileSync } from "node:fs"
import { join } from "node:path"

// The variables the start puts these in (`factorium_harness::harness::pi`).
const GUARD_ENV = "FACTORIUM_GUARD"
const WORKTREE_ENV = "FACTORIUM_WORKTREE"
const ADDED_ENV = "FACTORIUM_ADDED_DIR"
// Where a directory keeps its rules: the first of these it has, as pi reads
// the directory it starts in.
const RULES = ["AGENTS.md", "CLAUDE.md"]
// Where the added directory keeps its rules (`factorium_harness::start::RULES`).
const ADDED_RULES = ["CLAUDE.md"]
// How long the guard is given to answer one call, in milliseconds.
const GUARD_TIMEOUT = 60000

function fact(name) {
  const value = process.env[name]
  if (!value) throw new Error(`factorium: ${name} is not set: nothing is allowed unread`)
  return value
}

export default function (pi) {
  // The guard answers with its exit status: 0 allows the call, and anything
  // else — a refusal, a guard that is not there, one that did not answer in
  // time — refuses it. A handler that throws refuses the call too.
  pi.on("tool_call", (event, ctx) => {
    const described = JSON.stringify({ toolName: event.toolName, input: event.input ?? {} })
    const asked = spawnSync("/bin/sh", ["-c", fact(GUARD_ENV)], {
      input: described,
      cwd: ctx.cwd,
      encoding: "utf8",
      timeout: GUARD_TIMEOUT,
    })
    if (!asked.error && asked.status === 0) return undefined
    const said = (asked.stderr ?? "").trim()
    const failed = asked.error ? `: ${asked.error.message}` : ""
    return { block: true, reason: said || `factorium guard: it could not be asked${failed}` }
  })

  pi.on("before_agent_start", (event) => {
    const loaded = event.systemPromptOptions.contextFiles
    const kept = [
      [fact(WORKTREE_ENV), RULES],
      [fact(ADDED_ENV), ADDED_RULES],
    ]
    for (const [dir, names] of kept) {
      for (const name of names) {
        const path = join(dir, name)
        if (!existsSync(path)) continue
        if (!loaded.some((file) => file.path === path)) {
          loaded.push({ path, content: readFileSync(path, "utf8") })
        }
        break
      }
    }
  })
}

// No DSH imports: a plugin loaded by absolute filename cannot resolve them.
import { spawnSync } from 'node:child_process'
import { readFileSync, existsSync, readdirSync } from 'node:fs'
import { join } from 'node:path'
import http from 'node:http'
export const name = 'factorium'
export const inject = ['systemPrompt']

function fact(name) {
  const value = process.env[name]
  if (!value) throw new Error(`factorium: ${name} is missing; refusing DSH session`)
  return value
}

// The endpoint is a shell environment file. Read only its literal Orca
// assignments, never execute it or accept a remote HTTP destination.
function endpoint() {
  const env = { ...process.env }
  if (env.ORCA_AGENT_HOOK_ENDPOINT) {
    let text
    try { text = readFileSync(env.ORCA_AGENT_HOOK_ENDPOINT, 'utf8') }
    catch { return env }
    for (const line of text.split(/\r?\n/)) {
      const match = /^(?:export )?(ORCA_AGENT_HOOK_(?:PORT|TOKEN|ENV|VERSION))=(.*)$/.exec(line.trim())
      if (!match) continue
      const value = match[2].replace(/^(['"])(.*)\1$/, '$2')
      env[match[1]] = value
    }
  }
  return env
}

async function report(event, agent) {
  const env = endpoint()
  const port = Number(env.ORCA_AGENT_HOOK_PORT)
  if (!Number.isInteger(port) || port < 1 || port > 65535 || !env.ORCA_AGENT_HOOK_TOKEN) return
  const form = new URLSearchParams({
    paneKey: env.ORCA_PANE_KEY ?? env.ORCA_AGENT_PANE ?? '',
    launchToken: env.ORCA_AGENT_LAUNCH_TOKEN ?? env.ORCA_AGENT_LAUNCH ?? '',
    tabId: env.ORCA_TAB_ID ?? '', worktreeId: env.ORCA_WORKTREE_ID ?? '',
    env: env.ORCA_AGENT_HOOK_ENV ?? '', version: env.ORCA_AGENT_HOOK_VERSION ?? '',
    payload: JSON.stringify({ hook_event_name: event, session_id: agent?.id ?? '', cwd: process.cwd() }),
  }).toString()
  await new Promise(resolve => {
    const req = http.request({ hostname: '127.0.0.1', port, path: '/hook/dsh', method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded',
        'Content-Length': Buffer.byteLength(form), 'X-Orca-Agent-Hook-Token': env.ORCA_AGENT_HOOK_TOKEN },
    }, res => { res.resume(); res.on('end', resolve) })
    req.setTimeout(1500, () => req.destroy())
    req.on('error', resolve)
    req.end(form)
  })
}

function rules(dir, names) {
  for (const name of names) {
    const path = join(dir, name)
    if (existsSync(path)) return `\n${path}:\n${readFileSync(path, 'utf8')}\n`
  }
  return ''
}

function skills(dirs) {
  const named = new Map()
  for (const dir of dirs) {
    if (!existsSync(dir)) continue
    for (const entry of readdirSync(dir).sort()) {
      const path = join(dir, entry, 'SKILL.md')
      if (!existsSync(path) || named.has(entry)) continue
      named.set(entry, `${entry}: ${path}\n${readFileSync(path, 'utf8').split('---')[1] ?? ''}`)
    }
  }
  return '\nAvailable skills (read the named SKILL.md before applying one):\n' + [...named.values()].join('\n')
}

export function apply(ctx) {
  const guard = fact('FACTORIUM_GUARD')
  const worktree = fact('FACTORIUM_WORKTREE')
  const added = fact('FACTORIUM_ADDED_DIR')
  const home = process.env.FACTORIUM_DSH_KEPT || fact('DSH_HOME')
  const dir = process.cwd()
  ctx.on('tools/pre-execute', (exec, next) => {
    const asked = spawnSync('sh', ['-c', guard], {
      cwd: worktree, input: JSON.stringify({ name: exec.name, arguments: exec.arguments }),
      encoding: 'utf8', timeout: 60000,
    })
    if (asked.error || asked.status !== 0) return {
      kind: 'deny', reason: asked.stderr?.trim() || `factorium guard: ${asked.error?.message ?? 'could not be asked'}`,
    }
    return next()
  })
  // agent/status also fires for interrupted turns; turn-stopping does not.
  // Serialize reports and aggregate agents: one idle subagent cannot stop a
  // session whose parent or another subagent is still running.
  const running = new Set()
  let reports = Promise.resolve()
  const tell = (event, agent) => {
    reports = reports.catch(() => {}).then(() => report(event, agent))
    return reports
  }
  // The tool waits for tui-idle before sending the first brief. Creation
  // starts idle and emits no status transition, so announce that initial
  // state explicitly. A child created during a turn cannot mark it idle.
  ctx.on('agent/created', ({ agent }) => {
    if (running.size === 0) {
      tell('SessionStart', agent)
      return tell('Stop', agent)
    }
    return reports
  })
  ctx.on('agent/status', ({ agent, status }) => {
    const wasRunning = running.size > 0
    if (status === 'running') running.add(agent.id)
    else if (status === 'idle') running.delete(agent.id)
    const isRunning = running.size > 0
    if (wasRunning !== isRunning) {
      tell(isRunning ? 'UserPromptSubmit' : 'Stop', agent)
    }
    return reports
  })
  ctx.on('agent/disposed', ({ agent }) => {
    if (running.delete(agent.id) && running.size === 0) return tell('Stop', agent)
    return reports
  })
  const text = rules(join(home, 'agents'), ['AGENTS.md']) + rules(home, ['AGENTS.md']) +
    '\nThe brief, start directory, added and repo rules take precedence over host user rules.\n' +
    rules(dir, ['CLAUDE.md']) + rules(added, ['CLAUDE.md']) + rules(worktree, ['AGENTS.md', 'CLAUDE.md']) +
    skills([join(dir, '.claude/skills'), join(home, 'host-skills'), join(home, 'host-agents-skills')])
  ctx.systemPrompt.section({ name: 'factorium-rules', order: 0, text })
  ctx.on('agent/request', async (_event, next) => {
    const config = await next()
    const model = process.env.FACTORIUM_DSH_MODEL
    return model ? { ...config, model } : config
  })
  // Retain the middleware's continuation: ignoring it prevents every turn.
  ctx.on('agent/pre-step', (_event, next) => next())
  // Loader dependencies on this service keep the TUI and agent loop from
  // mounting if import, injection or apply failed. Disposal also withdraws
  // the service, so a plugin unload cannot leave an unguarded session running.
  ctx.provide('factoriumGuard', true)
}

#!/bin/sh
# Keep the host's launch command opaque and append DSH's arguments as words.
# The worktree is the workspace target and the brief the first prompt: a
# launch with a first prompt opens no first-run guide, launchpad or home.
# The native loader dependency, not this wrapper, enforces guard activation.
set -eu
launch=$1
kept=$2
role=$3
worktree=$4
profile=$5
brief=$6
for file in factorium.mjs factorium.patch.yml; do
    if [ ! -r "$kept/$file" ]; then
        echo "factorium: DSH guard file $file is missing; refusing session" >&2
        exit 1
    fi
done
set --
if [ "$profile" = kept ]; then
    set -- --patch "$kept/sandbox.patch.yml"
fi
if [ -f "$role/.dsh/cordis.patch.yml" ]; then
    set -- "$@" --patch "$role/.dsh/cordis.patch.yml"
fi
set -- "$@" --patch "$kept/factorium.patch.yml" "$worktree" "$brief"
exec sh -c "$launch \"\$@\"" factorium-dsh "$@"
